   114→function fmtValue(metric: string, value: number): string {
   115→  if (metric === "steps") return Math.round(value).toLocaleString("en-US");
   116→  if (metric === "distance" || metric === "spo2") return (Math.round(value * 10) / 10).toString();
   117→  if (metric === "sleep_duration") return (Math.round(value * 10) / 10).toString();
   118→  return Math.round(value).toLocaleString("en-US");
   119→}
